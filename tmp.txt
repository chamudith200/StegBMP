#define MESSEGE_LENGTH 32;

#pragma pack(push, 1)

int pxSize;
int PADDING;

// Bitmap
typedef struct BMP_Struct{
    // Header
    uint16_t Signature;
    uint32_t FileSize;
    uint32_t Reserved;
    uint32_t DataOffset;
}BMP_FileHeader;

typedef struct {
    // InfoHeader
    uint32_t Size;
    uint32_t Width;
    uint32_t Height;
    uint16_t Planes;
    uint16_t BitsPerPixel;
    uint32_t Compression;
    uint32_t ImageSize;
    uint32_t XpixelsPerM; 
    uint32_t YpixelsPerM;
    uint32_t ColorsUsed;
    uint32_t ImportantColors;
}BMP_InfoHeader;

#pragma pack(pop)

void init_BMP_Header(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader, int width, int height);
uint8_t *random_pic(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader);
int create_bmp(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader, uint8_t *pixelData, char *fileName);
uint8_t *encode_pic(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader, uint8_t *originalPixelData, char *messege, int size);
int is_odd(uint8_t byte);
uint8_t *read_bmp(char *fileName);
char *decode_bmp(BMP_InfoHeader *infoHeader, uint8_t *encodedPixelData);

int main(int argc, char **argv) {
    BMP_FileHeader bmp_FileHEader;
    BMP_InfoHeader bmp_InfoHeader;

    init_BMP_Header(&bmp_FileHEader, &bmp_InfoHeader, 0xFF, 0xFF);

void init_BMP_Header(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader, int width, int height) {
    // real file size with padding
    int padding;
    int fileSize;
    if (((width * 3) % 4 )== 0) {
        padding = 0;
    }
    else {
        padding = ((width * 3) / 4 + (width * 3) % 4) * 4 - (width * 3);
    }
    PADDING = padding;
    fileSize = ((width * 3) + padding) * height;
    pxSize = fileSize;
    
    // fileHeader
    fileHeader->Signature = 0x4d42; // "BM"
    fileHeader->FileSize = fileSize + sizeof(BMP_FileHeader) + sizeof(BMP_InfoHeader);
    fileHeader->Reserved = 0x0;
    fileHeader->DataOffset = sizeof(BMP_InfoHeader) + sizeof(BMP_FileHeader);

    // infoHeader
    infoHeader->Size = sizeof(BMP_InfoHeader);
    infoHeader->Width = width;
    infoHeader->Height = height;
    infoHeader->Planes = 1;
    infoHeader->BitsPerPixel = 24;
    infoHeader->Compression = 0;
    infoHeader->ImageSize = 0;
    infoHeader->XpixelsPerM = 0;
    infoHeader->YpixelsPerM = 0;
    infoHeader->ColorsUsed = 0;
    infoHeader->ImportantColors = 0;
}


// random picture make
uint8_t *random_pic(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader) {
    uint8_t *pixelData = malloc(pxSize);
    if (!pixelData) {
        printf("Failed to allocate memory!\n");
        return NULL;
    }
    int lineSize = infoHeader->Width*3 + PADDING;
    uint8_t B = 0xa0;
    uint8_t G = 0xff;
    uint8_t R = 0x0f;

    for (int y = 0; y < infoHeader->Height; y++) {
        for (int x = 0; x < infoHeader->Width; x++) {
            int index = y*lineSize + x*3;
            pixelData[index + 0] = B;
            pixelData[index + 1] = G;
            pixelData[index + 2] = R;
            B++;
            G--;
            R++;
        }
        // B = 0xa0;
        // G = 0xff;
        R = 0x0f;
    }

    return pixelData;
}


// Encode messege
uint8_t *encode_pic(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader, uint8_t *originalPixelData, char *messege, int size) {
    uint8_t *encodePixelData = malloc(pxSize);
    if (!encodePixelData) {
        printf("Failed to allocate memory!\n");
        return NULL;
    }
    
    infoHeader->XpixelsPerM = size;
    infoHeader->YpixelsPerM = size;
    memcpy(encodePixelData, originalPixelData, pxSize);

    int lineSize = infoHeader->Width * 3 + PADDING;
    for (int y = 0; y < size; y++) {
        char byte = messege[y];
        for (int x = 0; x < 8; x++) {
            int index = y*lineSize + x * 3;
            uint8_t bit = (byte >> (7 - x)) & 1;
            if (bit == 0b1) {
                if (!is_odd(encodePixelData[index])) {
                    encodePixelData[index] ^= 1 ;
                }
            }
            else {
                if (is_odd(encodePixelData[index])) {
                    encodePixelData[index] ^= 1;
                }
            }

        }
    }
    return encodePixelData;
}


int create_bmp(BMP_FileHeader *fileHeader, BMP_InfoHeader *infoHeader, uint8_t *pixelData, char *fileName) {
    FILE *fp_bmp = fopen(fileName, "wb");
    if (!fp_bmp) {
        printf("Failed to open %s\n", fileName);
        return 1;
    }
    
    // fileHeader
    fwrite(&fileHeader->Signature, sizeof(fileHeader->Signature), 1, fp_bmp);
    fwrite(&fileHeader->FileSize, sizeof(fileHeader->FileSize), 1, fp_bmp);
    fwrite(&fileHeader->Reserved, sizeof(fileHeader->Reserved), 1, fp_bmp);
    fwrite(&fileHeader->DataOffset, sizeof(fileHeader->DataOffset), 1, fp_bmp);

    // infoHeader
    fwrite(infoHeader, sizeof(*infoHeader), 1, fp_bmp);

    // PixelData
    fwrite(pixelData, pxSize, 1, fp_bmp);

    fclose(fp_bmp);
    printf("Created : %s\n", fileName);
    return 0;
}


int is_odd(uint8_t byte) {
    return byte % 2;
}


uint8_t *read_bmp(char *fileName) {
    BMP_FileHeader fileHeader;
    BMP_InfoHeader infoHeader;

    FILE *fp_bmp = fopen(fileName, "rb");
    if (!fp_bmp) {
        printf("Failed to open %s\n", fileName);
        return NULL;
    }
    
    uint8_t *pixelData = malloc(pxSize);
    if (!pixelData) {
        printf("Failed to allocate memory!\n");
        return NULL;
    }

    fread(&fileHeader, sizeof(BMP_FileHeader), 1, fp_bmp);
    fread(&infoHeader, sizeof(BMP_InfoHeader), 1, fp_bmp);

    fread(pixelData, fileHeader.FileSize - (sizeof(BMP_FileHeader) + sizeof(BMP_InfoHeader)), 1, fp_bmp);

    fclose(fp_bmp);
    return pixelData;
}


char *decode_bmp(BMP_InfoHeader *infoHeader, uint8_t *encodedPixelData) {
    if (!(infoHeader->XpixelsPerM & infoHeader->YpixelsPerM)) {
        printf("Encoding might be corrupted!\n");
        return NULL;
    }

    char *messege = malloc(sizeof(char) * infoHeader->XpixelsPerM);
    
    int lineSize = infoHeader->Width * 3 + PADDING;
    for (int y = 0; y < infoHeader->XpixelsPerM; y++) {
        uint8_t letter = 0x00;
        for (int x = 0; x < 8; x++) {   // x is the pixel offset not the byte offset
            int index = y*lineSize + x * 3;
            uint8_t bit = encodedPixelData[index] & 0b1;
            // if x2 becomes 0
            if (bit == 0b1) {
                // is a 1
                letter |= (0b1 << (7 - x));
            }
            // else {
            //     // is a 0
            //     bit |= !((0b1 << (7 - x)));
            // }
        }
        messege[y] = letter;
        printf("%i\n", letter);
    }
    return messege;
